19/// Represents a validated module package ready for installation.
20#[derive(Debug, Clone)]
21pub struct InstallPackage {
22    /// Base directory containing all package files (e.g., staging dir or original dir).
23    pub package_dir: Option<PathBuf>,
24
25    /// Path to the `.wasm` module inside `package_dir`.
26    pub wasm_path: PathBuf,
27
28    /// Path to the optional `module.component.toml` inside `package_dir`.
29    pub component_toml_path: Option<PathBuf>,
30
31    // В будущем:
32    // pub wit_path: Option<PathBuf>,
33    // pub license_path: Option<PathBuf>,
34}

36/// Validates that a given path points to a valid installable module package.
37///
38/// Rules:
39/// - Path must be a file with `.wasm` extension.
40/// - If the module is a WASI module (no WIT in .wasm), `component.toml` **must** exist.
41/// - If the module is a Component Model module (has WIT), `component.toml` is **optional**.
42///
43/// This function **does not parse** the contents — only checks file existence and naming.
44///
45/// Returns `Ok(InstallPackage)` if valid, or `Err` with clear diagnostics.
46pub async fn validate_install_package(wasm_path: &Path) -> ArcellaResult<InstallPackage> {
47    // 1. Проверка: это файл?
48    if !wasm_path.is_file() {
49        return Err(ArcellaError::InvalidArgument{
50            message: format!("Path is not a file: {:?}", wasm_path),
51        });
52    }
53
54    // 2. Проверка расширения
55    if wasm_path.extension().map_or(true, |ext| ext != "wasm") {
56        return Err(ArcellaError::InvalidArgument{
57            message: format!("File must have .wasm extension: {:?}", wasm_path),
58        });
59    }
60
61    // 3. Поиск '.component.toml' рядом
62    let stem = wasm_path.file_stem()
63        .ok_or_else(|| ArcellaError::InvalidArgument{
64            message: "Invalid filename".into(),
65        })?;
66    let expected_toml = wasm_path.with_file_name(format!("{}.component.toml", stem.to_string_lossy()));
67
68    let component_toml_path = if expected_toml.exists() {
69        Some(expected_toml)
70    } else {
71        None
72    };
73
74    // 4. (Опционально) можно уже здесь попытаться определить тип компонента
75    //    — но для MVP достаточно передать дальше
76
77    Ok(InstallPackage {
78        package_dir: None,
79        wasm_path: wasm_path.to_path_buf(),
80        component_toml_path,
81    })
82}

84/// Takes a validated package and stages it into a clean temporary subdirectory.
85///
86/// The subdirectory name is derived from the expected module ID (e.g., `name@version`).
87/// If the directory already exists, it is removed (with a warning) before staging.
88///
89/// Returns a new `InstallPackage` with all paths pointing inside the temp staging area.
90pub async fn prepare_install_package_in_temp(
91    storage: &StorageManager,
92    package: InstallPackage,
93) -> ArcellaResult<InstallPackage> {
94    let staging_dir = tempfile::tempdir_in(storage.temp_path())
95        .map_err(|e| ArcellaError::IoWithPath {
96            source: e,
97            path: storage.temp_path().to_path_buf(),
98        })?
99        .keep(); // Передаём владение в PathBuf
100
101    // Копирование .wasm
102    let wasm_file_name = package.wasm_path
103        .file_name()
104        .ok_or_else(|| ArcellaError::InvalidArgument { 
105            message: "WASM file has no name".into()
106        })?;
107    let staged_wasm = staging_dir.join(wasm_file_name);
108    tokio::fs::copy(&package.wasm_path, &staged_wasm).await
109        .map_err(|e| ArcellaError::IoWithPath {
110            source: e,
111            path: package.wasm_path.clone(),
112        })?;
113
114    // Копирование .component.toml (если есть)
115    let staged_toml = if let Some(ref src_toml) = package.component_toml_path {
116        let toml_file_name = src_toml.file_name()
117            .ok_or_else(|| ArcellaError::InvalidArgument {
118                message: "TOML file has no name".into()
119            })?;
120        let dst_toml = staging_dir.join(toml_file_name);
121        tokio::fs::copy(src_toml, &dst_toml).await
122            .map_err(|e| ArcellaError::IoWithPath {
123                source: e,
124                path: src_toml.clone(),
125            })?;
126        Some(dst_toml)
127    } else {
128        None
129    };
130
131    // Возвращаем новый пакет с путями внутри staging-директории
132    Ok(InstallPackage {
133        package_dir: Some(staging_dir.clone()), // Сохраняем путь к новой директории
134        wasm_path: staged_wasm,
135        component_toml_path: staged_toml,
136    })
137}

139/// Sanitizes a module ID for use as a filesystem directory name.
140///
141/// Replaces characters that are invalid or problematic in filenames:
142/// - '@' → '__at__'
143/// - '/' → '__slash__'
144/// - '\' → '__bslash__'
145/// - ':' → '__colon__'
146/// - и т.д.
147///
148/// WARNING: This is not cryptographically safe — only for local filesystem use.
149fn sanitize_module_id(id: &str) -> String {
150    id.replace('@', "__at__")
151        .replace('/', "__slash__")
152        .replace('\\', "__bslash__")
153        .replace(':', "__colon__")
154        .replace('?', "__qmark__")
155        .replace('*', "__star__")
156        .replace('|', "__pipe__")
157        .replace('<', "__lt__")
158        .replace('>', "__gt__")
159        .replace('"', "__quote__")
160}

162/// Ensures the final module directory does not exist in state or on disk.
163pub async fn check_module_not_installed(
164    state: &ArcellaState,
165    modules_dir: &Path,
166    module_id: &str,
167) -> ArcellaResult<()> {
168    if state.installed_modules.contains_key(module_id) {
169        let e = ArcellaError::ModuleAlreadyInstalled(module_id.to_string());
170        tracing::warn!("{}", e);
171        return Err(e);
172    }
173
174    let dest_dir = modules_dir.join(module_id);
175    if dest_dir.exists() {
176        let e = ArcellaError::ModuleDirAlreadyExists(module_id.to_string());
177        tracing::warn!("{}", e);
178        return Err(e);
179    }
180
181    Ok(())
182}

184/// Copies staged package files to the final module directory with durability guarantees.
185pub async fn install_module_files_to_storage(
186    staged: &InstallPackage,
187    modules_dir: &Path,
188    module_id: &str,
189) -> ArcellaResult<PathBuf> {
190    let dest_dir = modules_dir.join(module_id);
191    fs::create_dir_all(&dest_dir).await?;
192
193    // Copy .wasm
194    let wasm_file_name = staged.wasm_path
195        .file_name()
196        .ok_or_else(|| ArcellaError::InvalidArgument { 
197            message: "WASM file has no name".into()
198        })?;
199    let dest_wasm = dest_dir.join(wasm_file_name);
200    fs::copy(&staged.wasm_path, &dest_wasm).await?;
201    fs::File::open(&dest_wasm).await?.sync_all().await?;
202
203    // Copy .component.toml if present
204    if let Some(ref src_toml) = staged.component_toml_path {
205        let toml_file_name = src_toml.file_name()
206            .ok_or_else(|| ArcellaError::InvalidArgument {
207                message: "TOML file has no name".into()
208            })?;
209        let dest_toml = dest_dir.join(toml_file_name);
210        fs::copy(src_toml, &dest_toml).await?;
211        fs::File::open(&dest_toml).await?.sync_all().await?;
212    }
213
214    // Sync directory metadata (critical for durability of file names)
215    #[cfg(unix)]
216    std::fs::File::open(&dest_dir)?.sync_all()?;
217
218    Ok(dest_dir)
219}
