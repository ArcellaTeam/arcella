32/// Represents a validated module package ready for deployment.
33#[derive(Debug, Clone)]
34pub struct DeployPackage {
35    /// Base directory containing all package files (e.g., staging dir or original dir).
36    pub package_dir: Option<PathBuf>,
37
38    /// Path to the `.deployment.toml` file inside `package_dir`.
39    pub deployment_toml_path: PathBuf,
40
41    /// Extracted deployment ID (e.g., "web" from "web.deployment.toml").
42    pub deployment_id: String,
43
44}
45
46impl DeployPackage {
47    /// Validates that the deployment file exists and parses as a `DeploymentSpec`.
48    pub async fn parse_spec(&self) -> ArcellaResult<DeploymentSpec> {
49        DeploymentSpec::from_file(&self.deployment_toml_path)
50    }
51
52    /// Returns all existing file paths as owned `PathBuf`s.
53    ///
54    /// Only includes paths that are `Some`. The `.wasm` path is always included.
55    /// Order is deterministic but should not be relied upon externally.
56    pub fn existing_file_paths_owned(&self) -> Vec<PathBuf> {
57        let mut paths = Vec::with_capacity(1);
58        paths.push(self.deployment_toml_path.clone());
59
60        paths
61    }    
62
63    /// Reconstructs a `DeployPackage` from a staging directory.
64    ///
65    /// Assumes the deployment file has been copied into `staging_dir` with its original name.
66    pub fn from_staging_dir(&self, staging_dir: PathBuf) -> ArcellaResult<Self> {
67        let staged_path = staging_dir.join(
68            self.deployment_toml_path
69                .file_name()
70                .ok_or_else(|| ArcellaError::InvalidArgument {
71                    message: "Deployment file has no name".into(),
72                })?,
73        );
74
75        if !staged_path.exists() {
76            return Err(ArcellaError::InvalidArgument {
77                message: format!("Staged deployment file not found: {:?}", staged_path),
78            });
79        }
80
81        Ok(DeployPackage {
82            package_dir: Some(staging_dir),
83            deployment_toml_path: staged_path,
84            deployment_id: self.deployment_id.clone(),
85        })
86    }
87
88}

90/// Validates a user-provided `.deployment.toml` path.
91///
92/// Checks:
93/// - File exists and has `.deployment.toml` extension,
94/// - Prefix before `.deployment.toml` is a valid deployment ID,
95/// - Referenced `module_id` is syntactically valid.
96///
97/// Does **not** check if the module is installed — that happens later.
98pub async fn validate_deploy_package(deploy_path: &Path) -> ArcellaResult<DeployPackage> {
99    if !deploy_path.is_file() {
100        let e = ArcellaError::InvalidArgument{
101            message: format!("Path is not a file: {:?}", deploy_path),
102        };
103        tracing::error!("{}", e);
104        return Err(e);
105    }
106
107    // Extract and validate deployment ID
108    let deployment_id = match base_name_from_file_with_ext(deploy_path, "deployment.toml") {
109        Ok(id) => id,
110        Err(e) => {
111            tracing::error!("{}", e);
112            return Err(e);
113        }
114    };
115    match validate_base_name(&deployment_id) {
116        Ok(_) => (),
117        Err(e) => {
118            tracing::error!("{}", e);
119            return Err(e);
120        }
121    };
122
123    Ok(DeployPackage {
124        package_dir: None,
125        deployment_toml_path: deploy_path.to_path_buf(),
126        deployment_id: deployment_id.to_string(),
127    })
128
129}

131/// Stages the deployment file into a temporary directory and validates its contents.
132///
133/// Also ensures that the referenced module is **already installed** (in state or on disk).
134pub async fn prepare_deploy_package_in_temp(
135    storage: &StorageManager,
136    state: &ArcellaState,
137    package: DeployPackage,
138) -> ArcellaResult<DeployPackage> {
139    // Parse spec early to get module_id
140    let spec = package.parse_spec().await?;
141    let module_id = spec.module_id.to_string();
142
143    // Ensure module is installed
144    if !state.installed_modules.contains_key(&module_id) {
145        let e = ArcellaError::ModuleNotInstalled(module_id);
146        tracing::warn!("{}", e);
147        return Err(e);
148    }
149
150    // Stage file
151    let staging_dir = create_temp_subdir(storage.temp_path(), Some("deploy-staging")).await?;
152    let source_files = package.existing_file_paths_owned();
153    let _ = copy_files_to_dir(&source_files, &staging_dir, false).await?;
154
155    package.from_staging_dir(staging_dir)
156}
