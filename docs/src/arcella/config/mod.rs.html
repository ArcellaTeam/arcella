10//! Configuration loading, merging, and validation for the Arcella runtime.
11//!
12//! This module is responsible for:
13//! - Locating the base directory and configuration directory.
14//! - Ensuring the main configuration file (`arcella.toml`) exists (creating it from a template if needed).
15//! - Loading the built-in default configuration and user-provided configuration files recursively.
16//! - Merging configuration layers with strict rules for overriding and extension.
17//! - Validating the integrity of critical configuration files via modification time (`mtime`) checks.
18//!
19//! The configuration model enforces a layered approach:
20//! 1. **Built-in defaults** — minimal safe defaults shipped with Arcella.
21//! 2. **Main config (`arcella.toml`)** — user-editable root configuration.
22//! 3. **Included configs** — additional files loaded via `includes` directives.
23//!
24//! Only keys explicitly marked with `#redef` in a higher-priority layer may be overridden by lower layers.
25//! New keys may only be introduced under the `arcella.custom` or `arcella.modules` namespaces.
26

121/// Tracks file modification times to detect unauthorized changes after startup.
122#[derive(Debug, Clone)]
123pub struct IntegrityChecker {
124
125    /// List of paths to monitor.
126    paths: Vec<PathBuf>,
127
128    /// Initial modification times recorded at startup.
129    initial_mtimes: HashMap<PathBuf, SystemTime>,
130}
131
132impl IntegrityChecker {
133    /// Creates a new integrity checker by recording the current `mtime` of all given paths.
134    ///
135    /// **Note**: This function is synchronous and blocks the current thread.
136    /// For production use, it should be called via `tokio::task::spawn_blocking`.
137    pub fn new(paths: Vec<PathBuf>) -> ArcellaResult<Self> {
138        // TODO: Keep synchronous for MVP. In the future, consider making it fully asynchronous.
139        let mut initial_mtimes = HashMap::new();
140        for path in &paths {
141            let metadata = std::fs::metadata(path)
142                .map_err(|e| ArcellaError::IoWithPath { source: e, path: path.clone() })?;
143            let mtime = metadata.modified()
144                .map_err(|e| ArcellaError::Internal(format!("Cannot get mtime for {:?}: {}", path, e)))?;
145            initial_mtimes.insert(path.clone(), mtime);
146        }
147        Ok(IntegrityChecker { paths, initial_mtimes })
148    }
149
150    /// Checks whether any monitored file has been modified since startup.
151    ///
152    /// Returns an error if any file's `mtime` differs from the recorded value.
153    pub async fn check(&self) -> ArcellaResult<()> {
154        let current_mtimes = get_current_mtimes(&self.paths).await?;
155        check_mtimes_changed(&self.initial_mtimes, &current_mtimes)
156    }
157}

159/// Adds a warning that a value was ignored because the corresponding key
160/// was not marked with `#redef` in the main configuration layer.
161///
162/// # Arguments
163///
164/// * `key` – The configuration key that was ignored.
165/// * `config_idx` – The index of the config (in the `configs` vector) that lacked the `#redef` permission.
166/// * `config_files` – The global set of loaded config files (used to resolve file indices to paths).
167/// * `source_file_idx` – The index of the file that attempted to set the value (used for diagnostics).
168/// * `warnings` – Mutable vector to which the warning will be appended.
169fn add_no_redef_warning(
170    key: String,
171    config_idx: usize,
172    config_files: &IndexSet<PathBuf>,
173    source_file_idx: usize,
174    warnings: &mut Vec<fs_utils::ConfigLoadWarning>,
175) {
176    // Resolve the file path that attempted to set the value
177    let source_file_path = config_files
178        .get_index(source_file_idx)
179        .cloned()
180        .unwrap_or_else(|| PathBuf::from("unknown"));
181
182    warnings.push(fs_utils::ConfigLoadWarning::ValueError {
183        key,
184        error: format!(
185            "Value from file {:?} ignored due to no #redef flag in layer {}",
186            source_file_path, config_idx
187        ),
188        file: source_file_path,
189    });
190}

192/// Compares initial and current file modification times.
193///
194/// Returns an error if any file was modified or if a file is missing from the initial list.
195fn check_mtimes_changed(
196    initial_mtimes: &HashMap<PathBuf, std::time::SystemTime>,
197    current_mtimes: &HashMap<PathBuf, std::time::SystemTime>,
198) -> ArcellaResult<()> {
199    for (path, current_mtime) in current_mtimes {
200        if let Some(initial_mtime) = initial_mtimes.get(path) {
201            if current_mtime != initial_mtime {
202                return Err(ArcellaError::Internal(
203                    format!("Config integrity violation: file {:?} was modified after startup", path)
204                ));
205            }
206        } else {
207            return Err(ArcellaError::Internal(
208                format!("Config integrity violation: file {:?} not found in initial list", path)
209            ));
210        }
211    }
212    Ok(())
213}

215/// Retrieves current modification times for a list of paths concurrently.
216async fn get_current_mtimes(paths: &[PathBuf]) -> ArcellaResult<HashMap<PathBuf, std::time::SystemTime>> {
217    // Create a vector of futures for each mtime check
218    let checks: Vec<_> = paths.iter().map(|path| {
219        let path = path.clone();
220        async move {
221            let metadata = tokio::fs::metadata(&path).await
222                .map_err(|e| ArcellaError::IoWithPath { source: e, path: path.clone() })?;
223            let mtime = metadata.modified()
224                .map_err(|e| ArcellaError::Internal(format!("Cannot get mtime for {:?}: {}", path, e)))?;
225            Ok::<(PathBuf, std::time::SystemTime), ArcellaError>((path, mtime))
226        }
227    }).collect();
228
229    // Run all futures in parallel and wait for completion
230    let results = future::join_all(checks).await;
231
232    let mut current_mtimes = HashMap::with_capacity(results.len());
233    for result in results {
234        // If any check fails, propagate the error
235        let (path, mtime) = result?;
236        current_mtimes.insert(path, mtime);
237    }
238
239    Ok(current_mtimes)
240}

242/// Ensures that the main configuration file exists, creating it from a template if necessary.
243///
244/// Also ensures the template file (`arcella.template.toml`) exists.
245///
246/// **Note**: This function is not atomic and may be subject to race conditions
247/// if multiple Arcella instances start simultaneously. For production, consider
248/// using `O_CREAT | O_EXCL` semantics via `tokio::fs::OpenOptions::create_new(true)`.
249async fn ensure_main_config_exists(config_dir: &Path) -> ArcellaResult<(PathBuf, Vec<fs_utils::ConfigLoadWarning>)> {
250    let main_config_path = config_dir.join(MAIN_CONFIG_FILENAME);
251    let template_path = config_dir.join("arcella.template.toml");
252
253    let mut warnings: Vec<fs_utils::ConfigLoadWarning> = vec![];
254
255    // Create config_dir if it doesn't exist
256    fs::create_dir_all(config_dir)
257        .await
258        .map_err(|e| ArcellaError::IoWithPath { source: e, path: config_dir.to_path_buf() })?;
259
260    // Create template if missing
261    if !template_path.exists() {
262        fs::write(&template_path, TEMPLATE_CONFIG_CONTENT)
263            .await
264            .map_err(|e| ArcellaError::IoWithPath { source: e, path: template_path.clone() })?;
265        warnings.push(fs_utils::ConfigLoadWarning::Internal(
266            format!("Created default config template at {:?}", template_path)
267        ));
268    }
269
270    // Create main config from template if missing
271    if !main_config_path.exists() {
272        // TODO: Potential race condition if another process creates the file between `exists()` and `copy()`.
273        fs::copy(&template_path, &main_config_path).await?;
274        warnings.push(fs_utils::ConfigLoadWarning::Internal(
275            format!("Created default config at {:?}", main_config_path)
276        ));
277    }
278
279    Ok((main_config_path, warnings))
280
281}

283/// Intermediate representation of a resolved configuration value during merging.
284struct ResolvedValue {
285    value: TomlValue,
286    source_config_idx: usize,
287    source_file_idx: usize,          // index in `config_files`
288    redef_allowed_by: Option<usize>, // file index that granted redefinition permission
289}

291/// Loads and validates the complete Arcella configuration.
292///
293/// This is the main entry point for configuration initialization.
294/// It performs the following steps:
295/// 1. Locates the base directory.
296/// 2. Sets up the config directory and ensures `arcella.toml` exists.
297/// 3. Loads the built-in default configuration.
298/// 4. Recursively loads user configuration and included files.
299/// 5. Merges all layers according to Arcella's override rules.
300/// 6. Validates required paths and integrity.
301///
302/// Returns the final `ArcellaConfig` and any non-fatal warnings collected during loading.
303pub async fn load() -> ArcellaResult<(ArcellaConfig, Vec<fs_utils::ConfigLoadWarning>)> {
304    
305    // 1. Find base_dir
306    let base_dir = fs_utils::find_base_dir().await?;
307
308    // 2. Set config_dir
309    let config_dir = base_dir.join("config");    
310
311    // 3. Ensure config_dir and main config exist
312    let (main_config_path, warnings) = ensure_main_config_exists(&config_dir).await?;
313
314    // 4. Prepare paths for integrity checking (currently only main config)
315    let integrity_check_paths = vec![main_config_path.clone()];
316    let integrity_checker = IntegrityChecker::new(integrity_check_paths)?;
317    
318    // 5. Initialize loading state
319    let mut state  = fs_utils::ConfigLoadState {
320        config_files: IndexSet::new(),
321        visited_paths: HashSet::new(),
322        warnings,
323    };
324
325    // 6. Register built-in default config
326    let (file_idx, _) = state.config_files.insert_full(
327        PathBuf::from(DEFAULT_CONFIG_FILENAME)
328    );
329    let (default_config, _) = fs_utils::toml::parse_and_collect(
330        DEFAULT_CONFIG_CONTENT,
331        &vec!["arcella".to_string()],
332        file_idx,
333    )?;
334
335    // 7. Set up loading parameters
336    let params = fs_utils::ConfigLoadParams {
337        prefix: vec!["arcella".to_string()],
338        config_dir: config_dir.to_path_buf(),
339    };
340
341    // 8. Load main config and all included files recursively
342    let configs = fs_utils::load_config_recursive_from_file(
343        &params,
344        &mut state,
345        &main_config_path,
346    ).await?;
347
348    // 9. Merge configuration layers
349    let mut final_values = merge_config(
350        &default_config,
351        &configs,
352        &state.config_files,
353        &config_dir,
354        &mut state.warnings,
355    )?;
356
357    // 10. Sort keys for deterministic output
358    final_values.sort_keys();
359
360
361    integrity_checker.check().await?;
362
363    Ok((
364        ArcellaConfig {
365            config_values: final_values,
366            base_dir,
367            config_dir,
368            integrity_checker,
369        },
370        state.warnings,
371    ))
372}

374/// Helper to extract a required string path value from config by suffix.
375fn extract_path_value(config: &ConfigValues, suffix: &str) -> ArcellaResult<PathBuf> {
376    let full_key = format!("{}{}", ARCELLA_PREFIX, suffix);
377    match config.get(&full_key) {
378        Some((TomlValue::String(s), _)) => Ok(PathBuf::from(s)),
379        _ => Err(ArcellaError::Internal(format!("{} is not set or not a string", full_key))),
380    }
381}

383/// Merges configuration layers according to Arcella's strict override and extension rules.
384fn merge_config(
385    default_config: &fs_utils::TomlFileData,
386    configs: &Vec<fs_utils::TomlFileData>,
387    config_files: &IndexSet<PathBuf>,
388    config_dir: &Path,
389    warnings: &mut Vec<fs_utils::ConfigLoadWarning>
390) -> Result< ConfigValues, ArcellaError> {
391    
392    let mut preliminary_values: IndexMap<String, ResolvedValue> = IndexMap::new();
393
394    // Process from lowest to highest priority (i.e., reverse order of `configs`)
395    for config_idx in (0..configs.len()).rev() {
396        let config = &configs[config_idx];
397        for (key, (value, file_idx)) in &config.values {
398            // Check if the key ends with #redef
399            let (actual_key, is_redef) = if key.ends_with(REDEF_SUFFIX) {
400                // Extract the original key without the #redef suffix
401                (key[..key.len() - REDEF_SUFFIX.len()].to_string(), true)
402            } else {
403                (key.clone(), false)
404            };
405
406            match preliminary_values.entry(actual_key.clone()) {
407                Entry::Occupied(mut e) => {
408                    // Current layer has HIGHER priority (lower idx) than the existing entry
409                    if !is_redef { 
410                        add_no_redef_warning(
411                            actual_key.clone(),
412                            config_idx,
413                            config_files,
414                            e.get().source_file_idx ,
415                            warnings,
416                        );
417                        // Lower-priority layer is ignored; higher-priority layer sets the value.
418                        // Overwrite
419                        let e = e.get_mut();
420                        e.value = value.clone();
421                        e.source_config_idx = config_idx;
422                        e.source_file_idx = *file_idx;   
423                    } else {
424                        // The key `actual_key` was already defined in a lower-priority layer
425                        e.get_mut().redef_allowed_by = Some(*file_idx);
426                    }
427                }
428                Entry::Vacant(_) => {
429                    // No value for this key yet. Store the current value
430                    preliminary_values.insert(
431                        actual_key, 
432                        ResolvedValue {
433                            value: value.clone(),
434                            source_config_idx: config_idx,
435                            source_file_idx: *file_idx,   
436                            redef_allowed_by: None,
437                        }
438                    );
439                }
440            }
441
442        }
443    }
444
445    let main_idx = config_files.get_index_of(&config_dir.join(MAIN_CONFIG_FILENAME)).expect("Main config must be in config_files");
446    let default_idx = config_files.get_index_of(
447        &PathBuf::from(DEFAULT_CONFIG_FILENAME)
448    ).expect("Default config must be in config_files");
449
450    let mut final_values: ConfigValues = IndexMap::new();
451
452    // Initialize final config from default config
453    // Perform initial population from the default configuration
454    for (key, (value, file_idx)) in &default_config.values {
455        final_values.insert(
456            key.clone(), 
457            (value.clone(), default_idx)
458        );  
459    }
460
461    // Merge `preliminary_values` with the default config
462    // Overriding values from the initial population is allowed only if:
463    // - The source is the main config (`source_file == main_idx`), OR
464    // - The main config contains this key with the `#redef` suffix (`redef_allowed_by == main_idx`)
465    // New keys (not present in the default config) are allowed only under
466    // `CUSTOM_PREFIX_FULL`, `MODULES_PREFIX_FULL`  or `LOG_INTERNAL_PREFIX_FULL`
467    for (key, preliminary_value) in &preliminary_values {
468        // Flag indicating that the configuration section allows
469        // adding new keys not present in the default configuration
470        let is_newable = key.starts_with(CUSTOM_PREFIX_FULL) 
471            || key.starts_with(MODULES_PREFIX_FULL)
472            || key.starts_with(LOG_INTERNAL_PREFIX_FULL);
473        let new_value = &preliminary_value.value;
474        let insert_index = preliminary_value.source_config_idx;
475
476        match final_values.entry(key.clone()) {
477            Entry::Occupied(mut entry) => {
478                // This key exists in the default configuration
479                if preliminary_value.source_file_idx == main_idx {
480                    // This value comes from the main config, so it can override the default
481                    entry.insert(
482                        (new_value.clone(), preliminary_value.source_file_idx)
483                    );
484                } else if preliminary_value.redef_allowed_by == Some(main_idx) {
485                    // Redefinition was allowed by the main config, so override is permitted
486                    entry.insert(
487                        (new_value.clone(), preliminary_value.source_file_idx)
488                    );
489                } else {
490                    // To override a default value, the main config must mark the key with `#redef`
491                    add_no_redef_warning(
492                        key.clone(),
493                        0,
494                        config_files,
495                        preliminary_value.source_file_idx,
496                        warnings,
497                    );
498                }
499            }
500            Entry::Vacant(_) => {
501                // This key is absent from the default config, so check
502                // that the new key is added under `arcella.custom` or `arcella.modules`
503                if is_newable {
504                    final_values.insert(
505                        key.clone(), 
506                        (new_value.clone(), preliminary_value.source_file_idx)
507                    );
508                } else {
509                    // Adding new keys to this section is not allowed
510                    warnings.push(fs_utils::ConfigLoadWarning::ValueError {
511                        key: key.clone(),
512                        error: format!(
513                            "Value from layer {} ignored due to missing in default config",
514                            insert_index
515                        ),
516                        file: PathBuf::from(format!("layer_{}.toml", insert_index)),
517                    });
518                }
519            }
520        }
521    };
522
523    Ok(final_values)
524
525}

528/// Extracts a TOML subtree from the flat `ConfigValues` map under a given prefix.
529///
530/// For example, given:
531/// ```toml
532/// arcella.log.level = "debug"
533/// arcella.log.dir = "/tmp"
534/// arcella.server.port = 8080
535/// ```
536/// Calling `extract_subtree(config, "arcella.log")` returns:
537/// ```toml
538/// level = "debug"
539/// dir = "/tmp"
540/// ```
541///
542/// Nested keys (e.g., `arcella.cache.redis.host`) are converted into nested TOML tables.
543pub fn extract_subtree(config: &ConfigValues, prefix: &str) -> Table {
544    let mut table = Table::new();
545    let prefix_with_dot = format!("{}.", prefix);
546
547    for (key, (value, _)) in config {
548        if key.starts_with(&prefix_with_dot) {
549            let subkey = &key[prefix_with_dot.len()..];
550            if subkey.contains('.') {
551                insert_nested(&mut table, subkey, value.clone());
552            } else {
553                table.insert(subkey.to_string(), toml_value_to_toml(value));
554            }
555        }
556    }
557
558    table
559}
560
561/// Recursively inserts a value into a TOML table using a dot-separated path.
562///
563/// Example: `insert_nested(table, "a.b.c", value)` produces `{ a: { b: { c: value } } }`.
564fn insert_nested(table: &mut Table, path: &str, value: TomlValue) {
565    let parts: Vec<&str> = path.split('.').collect();
566    insert_nested_rec(table, &parts, value);
567}
568
569fn insert_nested_rec(table: &mut Table, parts: &[&str], value: TomlValue) {
570    if parts.len() == 1 {
571        table.insert(parts[0].to_string(), toml_value_to_toml(&value));
572    } else {
573        let key = parts[0].to_string();
574        let entry = table.entry(key).or_insert_with(|| toml::Value::Table(Table::new()));
575        if let toml::Value::Table(ref mut subtable) = entry {
576            insert_nested_rec(subtable, &parts[1..], value);
577        }
578        // Ignore type conflicts for MVP (e.g., if key already exists as non-table)
579    }
580}

582/// Converts the internal `TomlValue` enum to `toml::Value`.
583fn toml_value_to_toml(value: &TomlValue) -> toml::Value {
584    match value {
585        TomlValue::String(s) => toml::Value::String(s.clone()),
586        TomlValue::Integer(i) => toml::Value::Integer(*i),
587        TomlValue::Boolean(b) => toml::Value::Boolean(*b),
588        TomlValue::Float(OrderedFloat(f)) => toml::Value::Float(*f),
589        // Fallback for unsupported types (arrays, tables) — should not occur in MVP
590        _ => toml::Value::String(format!("{:?}", value)),
591    }
592}
